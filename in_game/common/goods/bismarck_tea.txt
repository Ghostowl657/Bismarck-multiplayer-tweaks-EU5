# EU5 1.4 uses these goods properties for the Columbian Exchange action,
# market checks, AI selection, and the situation's goods list.
INJECT:tea = {
	origin_in_old_world = yes
	custom_tags = { old_world_goods }

	location_potential = {
		# Preserve existing tea production in its native Old World locations.
		# Apply the former exchange action's planting rules at its destinations.
		trigger_if = {
			limit = {
				OR = {
					continent = continent:america
					continent = continent:oceania
					region = region:macaronesia_region
				}
			}
			OR = {
				vegetation = woods
				vegetation = forest
				vegetation = grasslands
				vegetation = farmland
			}
			OR = {
				climate = subtropical
				climate = oceanic
			}
			OR = {
				topography = plateau
				topography = hills
				topography = mountains
			}
			NOR = {
				region = region:east_coast_region
				region = region:west_coast_region
				region = region:great_plains_region
			}
		}
	}
}
